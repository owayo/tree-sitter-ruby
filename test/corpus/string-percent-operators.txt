========================================
typed percent after a string is an operator
========================================

"a"%q[x]
'a' %Q[x]
%q[a]%[x]
?a%q[x]

---

(program
  (binary
    left: (string (string_content))
    right: (element_reference object: (identifier) (identifier)))
  (binary
    left: (string (string_content))
    right: (element_reference object: (constant) (identifier)))
  (binary
    left: (string (string_content))
    right: (array (identifier)))
  (binary
    left: (character)
    right: (element_reference object: (identifier) (identifier))))

========================================
quoted strings concatenate after percent strings
========================================

%q[a] "b"
%Q[a] 'b'
?a "b"
puts %q[a] "b"

---

(program

    (string (string_content)))

    arguments: (argument_list
      (chained_string
        (string (string_content))
